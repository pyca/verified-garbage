use std::{hint::black_box, time::Instant};
use verified_garbage::kdf::scrypt::derive;
use verified_garbage_asm_x86_64::dispatch::{scrypt::vg_scrypt_auto, zeroize::vg_zeroize_auto};
#[derive(Clone)]
#[repr(align(64))]
struct Chunk([u8;128]);
fn chunks(n:usize)->Vec<Chunk>{let mut v=Vec::new(); v.try_reserve_exact(n).unwrap(); v.resize(n,Chunk([0;128])); v}
fn main(){
 let args:Vec<_>=std::env::args().collect(); let n:usize=args[1].parse().unwrap(); let r:usize=args[2].parse().unwrap();let p:usize=args[3].parse().unwrap();let reps:usize=args[4].parse().unwrap();
 let bulk=std::env::var("SCRYPT_WIPE_MODE").as_deref()==Ok("bulk");
 for sample in 0..reps {
 let mut out=[0u8;64];
 let start=Instant::now();
 let mut b=chunks(r*p);let mut v=chunks(r*n);let mut s=chunks(r+16);
 let alloc=start.elapsed().as_secs_f64();let start=Instant::now();
 // Same valid, disjoint, aligned buffers and validated parameter cases as the public API.
 unsafe{vg_scrypt_auto(b"password".as_ptr(),8,b"NaCl".as_ptr(),4,r,b.as_mut_ptr().cast(),r*p,v.as_mut_ptr().cast(),r*n,s.as_mut_ptr().cast(),r+16,out.as_mut_ptr(),64)};
 let kernel=start.elapsed().as_secs_f64();black_box(&out);let start=Instant::now();
 // Match Memory::drop, including a dispatcher call for each 128-byte chunk.
 if bulk { for buf in [&mut b,&mut v,&mut s] { unsafe {vg_zeroize_auto(buf.as_mut_ptr().cast(),std::mem::size_of_val(buf.as_slice()))} } } else { for c in b.iter_mut().chain(v.iter_mut()).chain(s.iter_mut()){unsafe{vg_zeroize_auto(c.0.as_mut_ptr(),128)}} }
 let wipe=start.elapsed().as_secs_f64();let start=Instant::now();drop((b,v,s));let free=start.elapsed().as_secs_f64();let start=Instant::now();
 let mut expected=[0u8;64];derive(b"password",b"NaCl",n as u64,r as u32,p as u32,usize::MAX,&mut expected).unwrap();
 let full=start.elapsed().as_secs_f64();assert_eq!(out,expected);
 println!("{n},{r},{p},{sample},{alloc:.9},{kernel:.9},{wipe:.9},{free:.9},{full:.9}");
 }
}
