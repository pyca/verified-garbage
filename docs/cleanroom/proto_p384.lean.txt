def asmOf (n : String) (c : Prog isa) : String :=
  s!"\n.globl {n}\n{n}:\n" ++ String.join ((Rust.assembly "call" (VG.X86_64.printer.function c)).map fun l => "  " ++ l ++ "\n")
def loopOf (n : String) (c : Prog isa) : String :=
  let ls := ((asmOf "BODY" c).splitOn "BODY:\n").getLastD ""
  let ls := String.intercalate "\n" ((ls.splitOn "\n").filter (fun (l : String) => l.trimAscii.toString != "ret"))
  s!"\n.globl {n}\n{n}:\n push rbx\n push rbp\n push r12\n push r13\n push r14\n push r15\n sub rsp,8\n mov rbx,rsi\n1:\n{ls}\n sub rbx,1\n jnz 1b\n add rsp,8\n pop r15\n pop r14\n pop r13\n pop r12\n pop rbp\n pop rbx\n ret\n"
def dbls (ts : List Reg) (top : Reg) : List Instr := (match ts with | t :: r => .alu .add t (.reg t) :: r.map fun t => .alu .adc t (.reg t) | _ => []) ++ [.mov32 top (.imm 0), .alu .adc top (.imm 0)]
def adds (ts : List Reg) (top : Reg) (a : Nat) : List Instr := chain .add .adc ts a ++ [.alu .adc top (.imm 0)]
def L6 := low 6
def T6 := top 6
def foldT : List Instr :=
  [.mov .rcx (.reg T6), .shift .shl .rcx 32, .alu .sub .rcx (.reg T6), .mov .rdx (.reg .rcx), .alu .add .rdx (.reg T6),
   .alu32 .xor .rax (.reg .rax), .alu .sub .rax (.reg .rcx), .alu .sbb .rdx (.imm 0)] ++
  [.alu .add .r8 (.reg .rax), .alu .adc .r9 (.reg .rdx), .alu .adc .r10 (.reg T6), .alu .adc .r11 (.imm 0), .alu .adc .r12 (.imm 0), .alu .adc .r13 (.imm 0),
   .mov32 T6 (.imm 0), .alu .adc T6 (.imm 0)]
def tri (M : Mod) (o a : Nat) : List Instr := loads L6 a ++ dbls L6 T6 ++ adds L6 T6 a ++ foldT ++ csub M L6 T6 ++ stores L6 o
def sub2 (M : Mod) (o a b : Nat) : List Instr := loads L6 M.mo ++ chain .sub .sbb L6 b ++ dbls L6 T6 ++ adds L6 T6 a ++ foldT ++ csub M L6 T6 ++ stores L6 o
def pr : List (Prog isa) → Prog isa | [] => .block [] | a :: l => .seq a (pr l)
def newd (M : Mod) (S : RcbSlots) (p : Pt) : Prog isa :=
  let F := fun x => opProg M x
  pr [F (.mul S.t0 p.z p.z), F (.mul S.t1 p.y p.y), F (.add S.t1 S.t1 S.t1),
   F (.mul S.t2 p.x S.t1), F (.sub S.t3 p.x S.t0), F (.add p.x p.x S.t0),
   F (.mul S.t3 S.t3 p.x), .block (tri M S.t3 S.t3),
   F (.add S.t2 S.t2 S.t2), F (.mul p.z p.y p.z), F (.add p.z p.z p.z),
   F (.mul p.x S.t3 S.t3), .block (sub2 M p.x p.x S.t2),
   F (.sub p.y S.t2 p.x), F (.mul p.y S.t3 p.y), F (.mul S.t1 S.t1 S.t1),
   .block (sub2 M p.y p.y S.t1)]
def K := (VG.Impl.P384.X86_64.publicJointAdx).K
def main : IO Unit := do
  IO.println ".intel_syntax noprefix\n.text"
  IO.println (asmOf "vg_p384_mul_mod_p_adx" (VG.Impl.Weierstrass.X86_64.Mont.mulFn6 true))
  IO.println (loopOf "p_old" (doubleIn K.M K.S K.R))
  IO.println (loopOf "p_new" (newd K.M K.S K.R))
  IO.println (loopOf "p_tri" (.block (tri K.M K.S.t0 K.S.t0)))
  IO.println (loopOf "p_two" (pr [opProg K.M (.add K.S.t1 K.S.t0 K.S.t0), opProg K.M (.add K.S.t0 K.S.t1 K.S.t0)]))
  IO.println (loopOf "p_mul" (opProg K.M (.mul K.S.t0 K.S.t0 K.S.t1)))
  IO.println s!"// mo={K.M.mo}"
